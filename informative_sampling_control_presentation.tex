\begin{frame}{From an optimal-control problem to a feedback policy}
Consider the constrained system
\[
  x_{k+1}=f(x_k,u_k),\qquad x_k\in\mathcal X,\quad u_k\in\mathcal U.
\]
At each measured state $x$, the controller solves the parametric problem
\begin{align*}
  z^\star(x)\in\arg\min_z\quad & J(z;x) \\
  \text{s.t.}\quad & c(z;x)=0, \qquad g(z;x)\le 0,
\end{align*}
where $z$ contains the predicted states and inputs. The first optimal input defines the expert feedback law
\[
  \pi^\star(x)=E z^\star(x)=u_0^\star(x).
\]

Repeated evaluation of this law produces the \textbf{expert closed loop}
\[
  x_{k+1}^\star=f\!\left(x_k^\star,\pi^\star(x_k^\star)\right).
\]
After training, the approximation $\pi_\theta$ defines a second closed loop
\[
  x_{k+1}^\theta=f\!\left(x_k^\theta,\pi_\theta(x_k^\theta)\right).
\]
\note{This slide introduces every object used later. A feedback policy maps the current state to the first optimal input. The phrase closed loop refers to the dynamics obtained after substituting that feedback law into the plant.}
\end{frame}

% -----------------------------------------------------------------------------
\begin{frame}{The dataset-design problem}
\small
Choose $q\in\mathcal P(\Xr)$, draw $N$ states, and fit the policy by empirical risk minimization:
\[
\begin{gathered}
  x_i\sim q,\qquad \D_N(q)=\{(x_i,\pi^\star(x_i))\}_{i=1}^{N},\\[0.25em]
  \widehat\theta(\D_N)
  \in\arg\min_\theta\frac1N\sum_{i=1}^{N}
  \|\pi_\theta(x_i)-\pi^\star(x_i)\|^2.
\end{gathered}
\]
For a horizon $T$, assume $x_k^\theta\in\Xr$ for $k<T$ and define the state distribution induced by the learned closed loop:
\[
  d_\theta^T=\frac1T\sum_{k=0}^{T-1}\operatorname{Law}(x_k^\theta),
  \qquad x_{k+1}^\theta=f(x_k^\theta,\pi_\theta(x_k^\theta)),\quad x_0\sim\mu_0.
\]
The closed-loop imitation risk and the resulting dataset-design problem are
\[
\begin{aligned}
  R_T(\theta)&=\mathbb E_{x\sim d_\theta^T}
  \|\pi_\theta(x)-\pi^\star(x)\|^2,\\[0.35em]
  q_N^\star&\in\arg\min_{q\in\mathcal P(\Xr)}
  \mathbb E_{\D_N(q)}\!\left[R_T\!\left(\widehat\theta(\D_N(q))\right)\right].
\end{aligned}
\]
\smallsource{The occupancy measure formalizes the distribution shift of sequential imitation learning: Ross, Gordon and Bagnell, AISTATS, 2011.}
\note{This is the precise outer problem. The sampling law determines the dataset, the dataset determines the trained parameters, the trained policy determines the visited-state distribution, and that distribution determines the relevant approximation error.}
\end{frame}

\begin{frame}{The feasible domain restricts the sampling law}
The OCP in slide 4 defines its $N$-step feasible set
\[
  \mathcal X_N=\left\{x\in\mathcal X:\exists z\ \text{such that}\ c(z;x)=0,\ g(z;x)\le0\right\}.
\]
For controller learning, choose a domain $\Xr\subseteq\mathcal X_N$ on which the exact MPC law is recursively feasible, and impose
\[
  q\in\mathcal P(\Xr),\qquad \operatorname{supp}(q)\subseteq\Xr.
\]
When $0<\operatorname{vol}(\Xr)<\infty$, the natural baseline is
\[
  q_{\mathrm{unif}}(x)=\frac{\mathbf 1_{\Xr}(x)}{\operatorname{vol}(\Xr)}.
\]
Uniform sampling covers volume, but the objective on slide 5 depends on how difficult $\pi^\star$ is to approximate and on which states $d_\theta^T$ visits.

\vspace{0.2cm}
\colorbox{PaleGold}{\parbox{0.9\textwidth}{\centering
$\Xr$ determines \textbf{where} labels are admissible. The remaining problem is \textbf{where inside $\Xr$} the $N$ labels should be placed.}}

\color{Slate}\scriptsize Recursive feasibility of the exact MPC law on $\Xr$ does not automatically transfer to an unconstrained neural approximation.
\note{This slide connects the sampling law q directly to the feasible set of the OCP. It also makes the important distinction between exact MPC recursive feasibility and the absence of an automatic guarantee for a plain neural policy.}
\end{frame}

\begin{frame}{Active-set geometry explains nonuniform difficulty}
Let $\mathcal A(x)$ denote the active inequality constraints and let $F_y$ denote the Jacobian of the reduced KKT system. Define a regular active-set region
\[
  \mathcal R_j=\left\{x\in\Xr:\mathcal A(x)=\mathcal A_j,\ F_y(y^\star(x),x)\ \text{nonsingular}\right\}.
\]
Under LICQ, strict complementarity, and a second-order sufficient condition, $z^\star(x)$ and $\pi^\star(x)$ are continuously differentiable in the interior of $\mathcal R_j$.

The critical set is
\[
  \mathcal B=Xr\setminus\bigcup_j\operatorname{int}(\mathcal R_j),
\]
which contains active-set transitions and points where the KKT regularity condition fails.

\begin{columns}[T,onlytextwidth]
\column{0.47\textwidth}
\textbf{Inside one region}

The policy is locally smooth. Sensitivities predict nearby labels and large uniform redundancy is possible.

\column{0.47\textwidth}
\textbf{Near $\mathcal B$}

The derivative can change discontinuously. Additional exact labels are needed to identify the neighboring regimes.
\end{columns}

\vspace{0.2cm}
\colorbox{PaleGold}{\parbox{0.9\textwidth}{\centering Slide 3 is the one-dimensional case: $\mathcal B=\{-0.5,0.5\}$.}}
\note{This slide provides the mathematical reason for nonuniform sampling. It does not claim that a nonlinear MPC law has a global finite polyhedral partition. The statement is local and conditional on standard parametric-NLP regularity assumptions.}
\end{frame}

% -----------------------------------------------------------------------------
\begin{frame}{KKT sensitivity gives local information from one solve}
\small
Fix $x_i\in\operatorname{int}(\mathcal R_j)$ and write the reduced KKT residual
\[
F(y,x)=
\begin{bmatrix}
\nabla_z L(z,\nu,\lambda;x)\\ c(z;x)\\ g_{\mathcal A}(z;x)
\end{bmatrix}=0,
\qquad y=(z,\nu,\lambda_{\mathcal A}).
\]

If $F_y(y_i^\star,x_i)$ is nonsingular, the implicit-function theorem gives
\[
  D_xy^\star(x_i)=-F_y^{-1}F_x,
  \qquad
  D_x\pi^\star(x_i)=E D_xz^\star(x_i).
\]
The first-order label at a nearby point is therefore
\[
  \widetilde\pi_i(x_i+\Delta x)
  =\pi^\star(x_i)+D_x\pi^\star(x_i)\Delta x.
\]
If $\pi^\star$ is twice differentiable on the segment and $\|D_x^2\pi^\star\|\le M_j$, Taylor's theorem yields the local error bound
\[
  \left\|\pi^\star(x_i+\Delta x)-\widetilde\pi_i(x_i+\Delta x)\right\|
  \le \frac{M_j}{2}\|\Delta x\|^2.
\]
\color{Slate}\small Thus, one exact OCP solve supplies a label and a locally valid model. The trust region must remain inside the same regular active-set region.

\vspace{0.08cm}{\color{Slate}\tiny Sensitivity-based augmentation: Krishnamoorthy, IEEE TAC 2022; predictor-corrector refinement: IEEE L-CSS 2023.}
\note{This slide now follows directly from slides 4 and 8. The KKT system belongs to the same parametric OCP, and its derivative produces a Taylor approximation of the same expert policy. The quadratic remainder bound states exactly when the augmented labels are reliable.}
\end{frame}

\begin{frame}{Research questions for a mathematical collaboration}
\begin{enumerate}
  \item[\color{Teal}\bfseries Q1.] \textbf{Definition.} Which information functional connects a sampling distribution to closed-loop loss, feasibility, or stability?

  \item[\color{Teal}\bfseries Q2.] \textbf{Guarantees.} Under piecewise-smooth parametric policies, can prior-informed sampling improve label complexity over uniform feasible-set sampling?

  \item[\color{Teal}\bfseries Q3.] \textbf{Geometry.} How should a sampler allocate mass near active-set boundaries, disconnected feasible components, and thin safety-critical regions?

  \item[\color{Teal}\bfseries Q4.] \textbf{Adaptation.} Can one control bias and variance when $q_t$ depends on the current learned policy and on previous optimization outcomes?
\end{enumerate}

\begin{frame}{A minimal first project}
\begin{columns}[T,onlytextwidth]
\column{0.61\textwidth}
\textbf{Benchmark}

A constrained nonlinear system such as cart-pole, followed by one switched-system or quadrupedal planning example.

\textbf{Four sampling baselines}
\begin{itemize}
  \item uniform box sampling with rejection,
  \item approximately uniform feasible-set sampling,
  \item active uncertainty sampling,
  \item the proposed structure-informed adaptive sampler.
\end{itemize}

\textbf{Fixed budget protocol}

Compare all methods for the same number of exact OCP solves, with identical network capacity and optimization settings.

\column{0.34\textwidth}
\textbf{Primary outcomes}
\begin{align*}
  &\text{closed-loop cost gap},\\
  &\text{constraint violation rate},\\
  &\text{feasible initial-state coverage},\\
  &\text{worst-case policy error},\\
  &\text{wall-clock data cost}.
\end{align*}

\vspace{0.15cm}
\colorbox{Sky}{\parbox{0.92\linewidth}{\centering First target: an empirical result plus one theorem in a tractable special case.}}
\end{columns}
\note{Keep the first project controlled. A linear or piecewise-affine case can support theory, while a nonlinear or switched example shows relevance to the user's research.}
\end{frame}

\begin{columns}[T,onlytextwidth]

\begin{frame}{Backup: differentiating the KKT system}
Let
\[
L(z,\nu,\lambda;x)=J(z;x)+\nu^\top c(z;x)+\lambda^\top g(z;x).
\]
For a fixed active set $\mathcal A$, the sensitivity solves
\[
\underbrace{
\begin{bmatrix}
\nabla^2_{zz}L & c_z^\top & g_{\mathcal A,z}^\top\\
c_z & 0 & 0\\
g_{\mathcal A,z} & 0 & 0
\end{bmatrix}}_{F_y}
\begin{bmatrix}
D_xz^\star\\D_x\nu^\star\\D_x\lambda^\star_{\mathcal A}
\end{bmatrix}
=-
\underbrace{
\begin{bmatrix}
\nabla^2_{zx}L\\c_x\\g_{\mathcal A,x}
\end{bmatrix}}_{F_x}.
\]

Local differentiability requires regularity conditions such as LICQ, strict complementarity, and a suitable second-order condition. Near active-set changes, generalized derivatives or predictor-corrector tracking may be more appropriate.
\note{Use this slide only if the audience asks where policy sensitivities come from. It also explains why active-set boundaries are statistically important and analytically difficult.}
\end{frame}
